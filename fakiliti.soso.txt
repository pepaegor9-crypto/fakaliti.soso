local function new(class, props, parent)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do i[k] = v end
    i.Parent = parent
    return i
end

local function corner(i, r) new("UICorner", {CornerRadius = UDim.new(0, r or 8)}, i) end
local function stroke(i, c, t, tr)
    return new("UIStroke", {
        Color = c or C.stroke,
        Thickness = t or 1,
        Transparency = tr or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, i)
end
local function grad(i, c1, c2, rot)
    return new("UIGradient", {Color = ColorSequence.new(c1, c2), Rotation = rot or 0}, i)
end
local function pad(i, t, r, b, l)
    new("UIPadding", {
        PaddingTop = UDim.new(0, t), PaddingRight = UDim.new(0, r),
        PaddingBottom = UDim.new(0, b), PaddingLeft = UDim.new(0, l),
    }, i)
end
local function tween(i, t, props, s, d)
    return SVC.TweenService:Create(i,
        TweenInfo.new(t, s or Enum.EasingStyle.Quad, d or Enum.EasingDirection.Out),
        props):Play()
end

local tabs, tabBtns, activeTab = {}, {}, nil

local function selectTab(name)
    for n, p in pairs(tabs) do p.Visible = (n == name) end
    for n, b in pairs(tabBtns) do
        if n == name then
            tween(b, 0.15, {BackgroundTransparency = 0})
            b.TextColor3 = C.text
            if b:FindFirstChild("UIStroke") then b.UIStroke.Color = C.accent end
        else
            tween(b, 0.15, {BackgroundTransparency = 0.55})
            b.TextColor3 = C.textDim
            if b:FindFirstChild("UIStroke") then b.UIStroke.Color = C.strokeDim end
        end
    end
    activeTab = name
end

local function makeTab(name, icon)
    local btn = new("TextButton", {
        Size = UDim2.new(1, 0, 0, 30),
        BackgroundColor3 = C.panel2,
        BackgroundTransparency = 0.55,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        Text = "  " .. icon .. "  " .. name,
        TextColor3 = C.textDim,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        AutoButtonColor = false,
        LayoutOrder = #tabBtns + 1,
    }, sidebar)
    corner(btn, 6)
    stroke(btn, C.strokeDim, 1, 0.5)
    btn.MouseEnter:Connect(function()
        if activeTab ~= name then tween(btn, 0.1, {BackgroundTransparency = 0.35}) end
    end)
    btn.MouseLeave:Connect(function()
        if activeTab ~= name then tween(btn, 0.1, {BackgroundTransparency = 0.55}) end
    end)
    btn.MouseButton1Click:Connect(function() selectTab(name) end)
    tabBtns[name] = btn
end

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; update(input)
        end
    end)
    SVC.UIS.InputChanged:Connect(function(input)
